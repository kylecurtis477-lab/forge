    fn new() -> Self {
        Self {

            vars: HashMap::new(),
            next_offset: -8,
            functions: HashMap::new(),
        }
    }

    fn new_label(&mut self) -> String {
        self.label_count += 1;
        format!("L{}", self.label_count)
    }

    fn get_var(&mut self, name: &str) -> i32 {
        if let Some(&o) = self.vars.get(name) {
            o
        } else {
            let o = self.next_offset;
            self.vars.insert(name.to_string(), o);
            self.next_offset -= 8;
            o
        }
    }

        // First pass: collect function definitions

// Constant folding for efficiency
fn fold(expr: &Expr) -> Expr {
    match expr {
        Expr::Binary(l, op, r) => {
            let fl = fold(l);
            let fr = fold(r);
            if let (Expr::Number(a), Expr::Number(b)) = (&fl, &fr) {
                let res = match op.as_str() {
                    "+" => a + b,
                    "-" => a - b,
                    "*" => a * b,
                    "/" => if *b != 0 { a / b } else { return Expr::Binary(Box::new(fl), op.clone(), Box::new(fr)); },
                    ">" => if a > b { 1 } else { 0 },
                    "<" => if a < b { 1 } else { 0 },
                    ">=" => if a >= b { 1 } else { 0 },
                    "<=" => if a <= b { 1 } else { 0 },
                    "==" => if a == b { 1 } else { 0 },
                    "!=" => if a != b { 1 } else { 0 },
                    _ => return Expr::Binary(Box::new(fl), op.clone(), Box::new(fr)),
                };
                return Expr::Number(res);
            }
            Expr::Binary(Box::new(fl), op.clone(), Box::new(fr))
        }
        Expr::Call(n, args) => Expr::Call(n.clone(), args.iter().map(fold).collect()),
        _ => expr.clone(),
    }
}

fn fold_stmts(stmts: &[Stmt]) -> Vec<Stmt> {
    stmts.iter().map(|s| match s {
        Stmt::Assign(n, e) => Stmt::Assign(n.clone(), fold(e)),
        Stmt::Print(e) => Stmt::Print(fold(e)),
        Stmt::If(c, t, e) => Stmt::If(fold(c), fold_stmts(t), e.as_ref().map(|eb| fold_stmts(eb))),
        Stmt::While(c, b) => Stmt::While(fold(c), fold_stmts(b)),
        Stmt::FuncDef(n, p, b) => Stmt::FuncDef(n.clone(), p.clone(), fold_stmts(b)),
        Stmt::Return(e) => Stmt::Return(e.as_ref().map(fold)),
        Stmt::ExprStmt(e) => Stmt::ExprStmt(fold(e)),
    }).collect()
}
